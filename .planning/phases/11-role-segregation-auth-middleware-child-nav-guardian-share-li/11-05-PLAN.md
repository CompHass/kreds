    Importações: NextRequest, NextResponse de 'next/server'; cookies de 'next/headers'; getChildSession de '@/lib/families/child-session'; db de '@/lib/db'; schema de '@/lib/db/schema'.

    Export async function POST(req, { params }):
    - const { childId } = await params
    - const cookieStore = await cookies()
    - const session = await getChildSession(cookieStore)
    - Se !session || session.childProfileId !== childId → return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

    Ler body: const { targetLabel, amountKreds } = await req.json().catch(() => ({}))
    Validar:
    - Se !targetLabel || typeof targetLabel !== 'string' || targetLabel.trim().length === 0 → return NextResponse.json({ error: 'targetLabel obrigatório' }, { status: 400 })
    - const amount = parseInt(String(amountKreds), 10); Se !Number.isInteger(amount) || amount < 1 → return NextResponse.json({ error: 'amountKreds deve ser inteiro positivo' }, { status: 400 })

    Insert:
    await db.insert(schema.donations).values({
      familyId: session.familyId,
      childProfileId: session.childProfileId,
      targetLabel: targetLabel.trim(),
      amountKreds: amount,
      status: 'pending',
    })

    Return NextResponse.json({ ok: true }, { status: 201 })
    Wrap com try/catch → 500 on error.
