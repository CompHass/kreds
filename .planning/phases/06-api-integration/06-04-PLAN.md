    3. Replace the existing `handleHarvest` function:
    ```
    async function handleHarvest() {
      if (harvestPending) return
      setHarvestPending(true)
      const totalAmount = tasks
        .filter(t => t.done)
        .reduce((sum, t) => sum + (t.kredsValue ?? 0), 0)

      try {
        const res = await fetch(`/api/child/${childId}/harvest`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ commandId: harvestCommandId, totalAmount, familyId: '' }),
        })
        if (res.ok) {
          setHarvested(true)
          setShowOverlay(true)
        } else if (res.status === 409) {
          // Already harvested — still show overlay (idempotent success)
          setHarvested(true)
          setShowOverlay(true)
        } else {
          console.error('Harvest failed', res.status)
        }
      } catch (e) {
        console.error('Harvest network error', e)
      } finally {
        setHarvestPending(false)
      }
    }
    ```

    NOTE about familyId in body: the harvest route handler ignores body.familyId and uses session.familyId (per Plan 06-03 security decision). Pass empty string `''` for now — the server reads familyId from the session cookie.
